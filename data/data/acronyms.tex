\chapter*{Kısaltmalar ve Terimler}
\addcontentsline{toc}{chapter}{Kısaltmalar ve Terimler}

% Sık kullanılan siber güvenlik kısaltmaları
\begin{description}[leftmargin=4.2cm,labelwidth=3.8cm,style=multiline]
  \item[CIA] Gizlilik (Confidentiality), Bütünlük (Integrity), Kullanılabilirlik (Availability)
  \item[AAA] Kimlik Doğrulama, Yetkilendirme, Hesap Verebilirlik (Accounting)
  \item[IAM] Kimlik ve Erişim Yönetimi (Identity and Access Management)
  \item[PAM] Ayrıcalıklı Erişim Yönetimi (Privileged Access Management)
  \item[SSO] Tekil Oturum (Single Sign-On)
  \item[MFA] Çok Faktörlü Kimlik Doğrulama (Multi-Factor Authentication)
  \item[RBAC] Rol Tabanlı Erişim Kontrolü (Role-Based Access Control)
  \item[ABAC] Öznitelik Tabanlı Erişim Kontrolü (Attribute-Based Access Control)
  \item[ACL] Erişim Kontrol Listesi (Access Control List)
  \item[DLP] Veri Kaybı Önleme (Data Loss Prevention)
  \item[EDR] Uç Nokta Tehdit Algılama ve Müdahale (Endpoint Detection and Response)
  \item[XDR] Genişletilmiş Algılama ve Müdahale (Extended Detection and Response)
  \item[SIEM] Güvenlik Bilgisi ve Olay Yönetimi (Security Information and Event Management)
  \item[SOAR] Güvenlik Orkestrasyonu, Otomasyon ve Yanıt (Security Orchestration, Automation and Response)
  \item[UEBA] Kullanıcı ve Varlık Davranış Analitiği (User and Entity Behavior Analytics)
  \item[NDR] Ağ Tabanlı Algılama ve Yanıt (Network Detection and Response)
  \item[ZTA/ZTNA] Sıfır Güven Mimarisi / Erişim Ağı (Zero Trust Architecture / Network Access)
  \item[CASB] Bulut Erişim Güvenlik Aracısı (Cloud Access Security Broker)
  \item[CSPM] Bulut Güvenliği Durum Yönetimi (Cloud Security Posture Management)
  \item[CWPP] Bulut İş Yükü Koruma Platformu (Cloud Workload Protection Platform)
  \item[CNAPP] Bulut Yerel Uygulama Koruma Platformu (Cloud-Native Application Protection Platform)
  \item[WAF] Web Uygulama Güvenlik Duvarı (Web Application Firewall)
  \item[RASP] Çalışma Zamanı Uygulama Öz Koruması (Runtime Application Self-Protection)
  \item[SAST] Statik Uygulama Güvenlik Testi (Static Application Security Testing)
  \item[DAST] Dinamik Uygulama Güvenlik Testi (Dynamic Application Security Testing)
  \item[IAST] Etkileşimli Uygulama Güvenlik Testi (Interactive Application Security Testing)
  \item[SCA] Yazılım Bileşen Analizi (Software Composition Analysis)
  \item[SBOM] Yazılım Malzeme Listesi (Software Bill of Materials)
  \item[CVE] Ortak Güvenlik Açığı Tanımlayıcısı (Common Vulnerabilities and Exposures)
  \item[CVSS] Ortak Zafiyet Puanlama Sistemi (Common Vulnerability Scoring System)
  \item[CWE] Ortak Zayıflık Sayımı (Common Weakness Enumeration)
  \item[ATT\&CK] MITRE Saldırı Bilgi Tabanı (Adversarial Tactics, Techniques and Common Knowledge)
  \item[TTP] Taktikler, Teknikler ve Prosedürler (Tactics, Techniques and Procedures)
  \item[IOC] İhlal Göstergesi (Indicator of Compromise)
  \item[IOA] Saldırı Göstergesi (Indicator of Attack)
  \item[CTI] Siber Tehdit İstihbaratı (Cyber Threat Intelligence)
  \item[IR] Olay Müdahale (Incident Response)
  \item[DFIR] Dijital Adli Bilişim ve Olay Müdahale (Digital Forensics and Incident Response)
  \item[SOC] Güvenlik Operasyon Merkezi (Security Operations Center)
  \item[NOC] Ağ Operasyon Merkezi (Network Operations Center)
  \item[GRC] Yönetişim, Risk ve Uyumluluk (Governance, Risk and Compliance)
  \item[ISO 27001] Bilgi Güvenliği Yönetim Sistemi Standardı
  \item[NIST CSF] NIST Siber Güvenlik Çerçevesi (Cybersecurity Framework)
  \item[NIST RMF] NIST Risk Yönetim Çerçevesi (Risk Management Framework)
  \item[CIS] Internet Security Center (CIS Kontrolleri)
  \item[HIPAA] Sağlık Sigortası Taşınabilirlik ve Sorumluluk Yasası (ABD)
  \item[GDPR] Genel Veri Koruma Tüzüğü (AB)
  \item[PCI DSS] Ödeme Kartı Endüstrisi Veri Güvenliği Standardı
  \item[OWASP] Açık Web Uygulama Güvenliği Projesi (Top 10 vb.)
  \item[STRIDE] Tehdit Modelleme Metodu: Spoofing, Tampering, Repudiation, Information Disclosure, Denial of Service, Elevation of Privilege
  \item[DREAD] Risk Puanlama: Damage, Reproducibility, Exploitability, Affected Users, Discoverability
  \item[KMS] Anahtar Yönetim Sistemi (Key Management System)
  \item[HSM] Donanım Güvenlik Modülü (Hardware Security Module)
  \item[PKI] Açık Anahtar Altyapısı (Public Key Infrastructure)
  \item[TLS] Aktarım Katmanı Güvenliği (Transport Layer Security)
  \item[DNSSEC] DNS Güvenlik Uzantıları (DNS Security Extensions)
  \item[MITM] Ortadaki Adam Saldırısı (Man-in-the-Middle)
\end{description}
